   id    : キャラごとの固有ID（半角英数字。一度決めたら変更しない）
   en    : 並び順（A〜Z）に使う英語名

   ※ 通常の技の「ヒート中版」「ヒートダッシュ版」は、元の技の直後に置き、t: "-ho" を付けてください（ヒート専用技の色が付かなくなります）。

     攻撃判定・コマンド・技名から自動で判定します。判定を直したいときは、その技に t を足します。

   ※ タグ列（技名の右）：カプセルのタグは、技名の欄ではなく右隣の「タグ」列に出ます（攻撃判定だけは技名の欄）。

const STORE_KEY = "tekken-checksheet-v1";
const LAST_KEY = "tekken-checksheet-last";

let state = {};            // { "char01:0:startup": true, ... }
let canStore = true;
let current = (CHARACTERS.find(c => c.moves.length) || CHARACTERS[0]).id;

function load() {
  try {
    state = JSON.parse(localStorage.getItem(STORE_KEY)) || {};
    const last = localStorage.getItem(LAST_KEY);
    if (last && CHARACTERS.some(c => c.id === last)) current = last;
  } catch (e) {
    canStore = false;
    document.getElementById("warn").style.display = "block";
  }
}
function save() {
  if (!canStore) return;
  try {
    localStorage.setItem(STORE_KEY, JSON.stringify(state));
    localStorage.setItem(LAST_KEY, current);
  } catch (e) {
    canStore = false;
    document.getElementById("warn").style.display = "block";
  }
}

/* 技の固定ID。k が無い技は並び順で代用し、同じキャラ内の重複は警告する */
const moveId = (raw, idx) => (raw && typeof raw === "object" && Number.isInteger(raw.k)) ? raw.k : idx;
CHARACTERS.forEach(c => {
  c.ks = new Set();
  c.moves.forEach((raw, idx) => {
    const k = moveId(raw, idx);
    if (c.ks.has(k)) console.error("技IDが重複しています: " + c.id + " k=" + k + " " + (raw && raw.name));
    c.ks.add(k);
  });
});

function normQ(t) {
  return String(t).toLowerCase().replace(/[\s　]+/g, "")
    .replace(/[Ａ-Ｚａ-ｚ０-９]/g, ch => String.fromCharCode(ch.charCodeAt(0) - 0xFEE0));
}
let reviewOnly = false;   // 復習モード：ブックマークの旗が付いた技だけを表示（保存しない）
let rowNums = [];         // 表の各行（技の位置 idx）の数字（数値の絞り込み・並べ替え用）
let rowCols = [];         // 表の各行（技の位置 idx）の、項目ごとの文字（ワード検索で、当たったセルに色を塗るため）
let nfApi = null;         // 絞り込みパネル（キャラを替えたら、状態の選択肢の「該当数」を作り直す）
function applyFilter() {
  const raw = document.getElementById("q").value || "";
  const nm = numMode(raw), q = nm.textOn ? normQ(raw) : "", hk = nm.nq ? "q" : "qa";   // 数字だけの検索は、絞り込まずにジャンプリンクを出す（文字での絞り込みに切り替えたときだけ絞り込む）。数字だけではない検索（ワード検索）は、技表の文字すべて（qa）が対象。数字だけの検索を文字で絞り込むときは、これまでどおり技名・コマンドなど（q）
  let total = 0, shownN = 0;
  document.querySelectorAll("#body tr[data-q]").forEach(tr => {
    total++;
    const hide = (!!q && !tr.dataset[hk].includes(q)) || !nfPass(rowNums[+tr.dataset.mi]) || (reviewOnly && !rvHas(current, +tr.dataset.k));
    tr.classList.toggle("nomatch", hide);
    if (!hide) shownN++;
  });
  applyOrder();
  const shown = new Set([...document.querySelectorAll("#body tr[data-mi]:not(.nomatch)")].map(r => +r.dataset.mi));
  jumpAvail(i => shown.has(i));   // 絞り込みで該当が無くなったジャンプリンクは薄くする
  numJumpRefresh(raw, [...document.querySelectorAll("#body tr[data-mi]")]
    .filter(tr => nfPass(rowNums[+tr.dataset.mi]) && !(reviewOnly && !rvHas(current, +tr.dataset.k)))
    .map(tr => ({ idx: +tr.dataset.mi, nums: rowNums[+tr.dataset.mi], hay: tr.dataset.q, vis: !tr.classList.contains("nomatch") })));   // 数字ジャンプ（表示の順）
  shApply();   // 検索に一致した所の色塗り・右端の目印
  updateFbar(shownN, total);
  scheduleFit(true);
}
/* 並べ替え（数値）：行の並びだけを入れ替える（行そのものは作り直さない）。並べ替えなしに戻したら、技表の順に戻す */
function applyOrder() {
  const body = document.getElementById("body");
  const sig = NF.sortCol ? NF.sortCol + "|" + NF.sortDir : "";
  if ((body.dataset.ord || "") === sig) return;
  const rows = [...body.querySelectorAll("tr[data-mi]")];
  nfSorted(rows, tr => rowNums[+tr.dataset.mi] || {}, tr => +tr.dataset.mi).forEach(tr => body.appendChild(tr));
  body.dataset.ord = sig;
  document.getElementById("jumpNav").classList.toggle("sorted", !!NF.sortCol);   // 並べ替え中は、技表の順に飛ぶジャンプリンクは使えない
  document.getElementById("jumpNav").title = NF.sortCol ? "並べ替え中はジャンプリンクを使えません（並べ替えなしに戻すと使えます）" : "";
}
function updateFbar(shownN, total) {
  const c = CHARACTERS.find(x => x.id === current), n = c ? rvCount(c) : 0;
  const rb = document.getElementById("rvTgl");
  rb.setAttribute("aria-pressed", String(reviewOnly));
  rb.classList.toggle("act", reviewOnly);
  rb.querySelectorAll(".lbl").forEach(e => { e.textContent = "ブックマークのみ表示：" + (e.dataset.v === "on" ? "ON" : "OFF") + "（" + n + "）"; });
  document.getElementById("rvClear").hidden = n === 0;
  const nb = document.getElementById("nfTgl");
  nb.classList.toggle("act", nfFilterOn() || nfSortOn());
  document.getElementById("nfTglT").textContent = nfBtnText();
  let t = "";
  if (total && (reviewOnly || nfFilterOn() || nfSortOn())) {
    t = "全" + total + "技中" + shownN + "技を表示";
    const miss = nfNoData(rowNums);
    if (miss.length) t += "（このキャラには「" + miss.join("」「") + "」の数字が登録されていません）";
    else if (reviewOnly && !n) t += "（技の行の旗マークでブックマークを付けると、ここに集まります）";
  }
  NF.stat = t;
  const fs = document.getElementById("fStat");
  fs.textContent = t; fs.hidden = !t;
  nfStatRefresh();
}

/* ---------- 技名セルを固定の高さに収める ----------
   行の高さは全部同じ（--row-h）。技名＋ふりがなが枠に収まらないときは文字を少しずつ縮め（--ts）、
   コマンドが1行に収まらないときは縮める（--cs）。1行で CMD_MIN 倍までしか縮まないものは、2行に折り返して大きく見せる。
   タグ列のカプセルは縮めない（全部同じ幅・同じ大きさで1段）。 */
let fitQueued = false, fitFull = false;
function scheduleFit(inc) {   // inc が true のとき（検索・絞り込みで呼ばれたとき）は、すでに収めてある行は測り直さない。それ以外（画面の大きさが変わった・文字が読み込まれた・表示項目を替えた）は全部やり直す
  if (inc !== true) fitFull = true;
  if (fitQueued) return;
  fitQueued = true;
  requestAnimationFrame(() => { fitQueued = false; const full = fitFull; fitFull = false; fitNames(!full); });
}
function fitKey(n) {   // 収め方は、枠の大きさと中身（検索で色を塗った文字・枠で囲んだコマンドも含む）で決まる。どれも同じなら、前に収めた結果をそのまま使える
  const c = n.querySelector(".mcmd");
  return n.clientWidth + "x" + n.clientHeight + "|" + (c && c.classList.contains("shl-cmd") ? 1 : 0) + "|" + n.firstChild.innerHTML;
}
const FIT_CMD_MIN = 0.8, FIT_CMD2 = 0.87;   // コマンドを1行で縮めてよい下限（12px の 0.8 倍＝約9.6px）／2行に折り返すときの大きさ
function fitNames(inc) {
  let list = [...document.querySelectorAll("#body tr:not(.nomatch) .nm")];
  if (!list.length || !list[0].getClientRects().length) return;   // 隠れている（コンボ集を表示中）ときは測れないので、技チェックシートに戻ったときにやり直す
  const keys = list.map(fitKey);
  if (!inc) document.querySelectorAll("#body tr.nomatch .nm").forEach(n => { n._fk = ""; });   // 隠れている行は今回測れないので、あとで表示されたときに測り直す
  if (inc) { const keep = []; list.forEach((n, i) => { if (n._fk !== keys[i]) keep.push(i); }); if (!keep.length) return; list = keep.map(i => list[i]); keep.forEach((i, j) => { keys[j] = keys[i]; }); keys.length = keep.length; }
  list.forEach((n, i) => { n._fk = keys[i]; });
  const tgOf = n => { const r = n.closest("tr"); return r ? r.querySelector(".tgc") : null; };
  list.forEach(n => {
    n.style.removeProperty("--ts"); n.title = "";
    const c = n.querySelector(".mcmd"); if (c) { c.style.removeProperty("--cs"); c.classList.remove("w2"); }
    const mi = n.querySelector(".minfo"); if (mi) mi.style.removeProperty("--is");
    const g = tgOf(n); if (g) { g.style.removeProperty("--tgs"); g.title = ""; }
  });
  // 読み取りをまとめて行う（書き込みと交互にしない）
  const cmds = [];
  list.forEach(n => {
    const c = n.querySelector(".mcmd");
    if (c && c.scrollWidth > c.clientWidth + 1) cmds.push({ c, v: Math.floor(c.clientWidth / c.scrollWidth * 96) / 100, two: false });
  });
  const infos = [];
  list.forEach(n => { const mi = n.querySelector(".minfo"); if (mi && mi.scrollWidth > mi.clientWidth + 1) infos.push({ mi, n, v: Math.floor(mi.clientWidth / mi.scrollWidth * 96) / 100 }); });
  infos.forEach(o => o.mi.style.setProperty("--is", o.v));
  let pi = infos;
  for (let k = 0; k < 6 && pi.length; k++) {
    pi = pi.filter(o => o.mi.scrollWidth > o.mi.clientWidth + 1);
    pi.forEach(o => { o.v = Math.max(0.5, Math.floor(o.v * 92) / 100); o.mi.style.setProperty("--is", o.v); });
  }
  pi.forEach(o => { o.n.title = infoShownText(o.mi); });
  cmds.forEach(o => {
    if (o.v < FIT_CMD_MIN) { o.two = true; o.v = Math.min(FIT_CMD2, Math.floor(o.v * 2 * 90) / 100); o.c.classList.add("w2"); }   // 1行だと小さくなりすぎるので、2行に折り返す
    o.c.style.setProperty("--cs", o.v);
  });
  let pc = cmds;
  for (let k = 0; k < 8 && pc.length; k++) {
    pc = pc.filter(o => o.two ? (o.c.scrollHeight > o.c.clientHeight + 1 || o.c.scrollWidth > o.c.clientWidth + 1) : o.c.scrollWidth > o.c.clientWidth + 1);
    pc.forEach(o => { o.v = Math.max(0.4, Math.floor(o.v * 93) / 100); o.c.style.setProperty("--cs", o.v); });
  }
  // コマンドの行数が決まってから、技名＋ふりがなが収まるかを測る
  let pending = list.filter(n => n.firstChild.scrollHeight > n.firstChild.clientHeight + 1);
  for (const s of [0.93, 0.87, 0.82, 0.77, 0.72, 0.66]) {
    if (!pending.length) break;
    pending.forEach(n => n.style.setProperty("--ts", s));
    pending = pending.filter(n => n.firstChild.scrollHeight > n.firstChild.clientHeight + 1);
  }
  pending.forEach(n => { n.title = n.firstChild.textContent; });   // どうしても入らないものは、ポインターを乗せると全文が出る
  // タグ列：カプセルは、全部同じ幅・同じ文字の大きさ・1段（縮めない）。もし枠に収まらない行があれば、ポインターを乗せると全部が出る
  list.map(tgOf).filter(g => g && g.scrollHeight > g.clientHeight + 1).forEach(g => { g.title = infoShownText(g); });
}

const charBtns = {};   // キャラid → 一覧のボタン（ブックマークの数の表示を更新するため）
function rvBadge(cid) {
  const c = CHARACTERS.find(x => x.id === cid);
  if (!c) return;
  const n = rvCount(c);
  [charBtns[cid], chdBtns[cid]].forEach(b => {   // 本体の一覧と、ドロワーの一覧の両方
    if (!b) return;
    if (n) { b.dataset.rv = n; b.title = c.name + "（ブックマーク" + n + "件）"; } else { delete b.dataset.rv; b.title = c.name; }
  });
}
function noDragImg(im) {   // 画像のドラッグ＆ドロップ（別の場所へ落とす・保存する）を無効にする
  im.draggable = false;
  im.addEventListener("dragstart", e => e.preventDefault());
  return im;
}
/* キャラの絵（一覧・ドロワーの顔・トップの絵）は、右クリックや長押しの「画像をコピー」「画像を保存」、ドラッグ、選択してのコピーで取り出せないようにする。
   絵に当たる操作（右クリック・長押しのメニュー・ドラッグ）を止め、全部選んでコピーしても、絵は入れず、文字だけをコピーする。表や文字のコピー、ふつうの右クリックは止めない */
const PIC_SEL = ".pimg, .chd-th, .chr";
function inPic(e) {
  const p = e.composedPath ? e.composedPath() : [e.target];
  return p.some(n => n && n.nodeType === 1 && n.matches && n.matches(PIC_SEL));
}
["contextmenu", "dragstart"].forEach(t => document.addEventListener(t, e => { if (inPic(e)) e.preventDefault(); }, true));
document.addEventListener("copy", e => {
  if (!e.clipboardData || !window.getSelection) return;
  const s = window.getSelection();
  if (!s || s.isCollapsed || !s.rangeCount) return;
  const touch = n => { for (let i = 0; i < s.rangeCount; i++) { try { if (s.getRangeAt(i).intersectsNode(n)) return true; } catch (x) {} } return false; };   // 選択の範囲に絵が入っているか（絵は選択できない設定なので、containsNode ではなく範囲そのもので調べる）
  const hit = [...document.querySelectorAll(".pimg img, .chd-th img, #hoHost")].some(touch);   // 一覧・ドロワーの顔、トップの大きな絵（#hoHost の中）
  if (hit) { e.preventDefault(); e.clipboardData.setData("text/plain", String(s)); }
}, true);

function renderCmd(el, str) {
  el.textContent = "";
  let last = 0, m, prevEnd = -1;
  const isLatin = ch => /[A-Za-z]/.test(ch || "");
  const text = t => { if (t) { const s = document.createElement("span"); s.className = "ct"; s.textContent = t; el.appendChild(s); } };
  CMD_RE.lastIndex = 0;
  while ((m = CMD_RE.exec(str))) {
    const s = m.index, e = s + m[0].length, prev = str[s - 1] || "", next = str[e] || "";
    if (m[0] === ".") { if (prevEnd === s && /[A-Za-z1-9]/.test(next)) { text(str.slice(last, s)); last = e; } continue; }
    if (m[2]) {
      if (next && next.charCodeAt(0) > 127 && next !== "）" && next !== "】" && next !== "【") continue;
      if (/レベル$/.test(str.slice(0, s))) continue;
    } else if (m[0] === "n") {
      if (isLatin(prev) || isLatin(next)) continue;
    }
    text(str.slice(last, s)); last = e; prevEnd = e;
    if (m[1]) el.appendChild(cmdIcon(CMD_COMBO[m[1]], CMD_LABEL[m[1]] || m[1]));
    else if (m[2]) {
      let d = m[2];
      if (d === "5") el.appendChild(cmdIcon("nn", "n"));
      else el.appendChild(cmdIcon((m[3] ? "h" : "d") + d, d + (m[3] ? "(ホールド)" : "")));
    } else el.appendChild(cmdIcon("nn", "n"));
  }
  text(str.slice(last));
}

  const still = /気合|挑発|礼|残心/.test(name || "") || /^特殊/.test(atk)                   // 攻撃のない動作
    || (su >= 30)                                                                           // 発生30F以上の溜め技・ガード不能技は1回の入力
    || SHEET_SIMUL.indexOf(name) >= 0;                                                      // 同時押しと判断した技（部分版が同じコマンドで存在するもの）
  const heat = /ヒート(?:発動可能)?状態で|ヒート中/.test(str);                              // ヒートバースト・ヒートスマッシュ（2+3 の同時押し）
  const seqPair = SHEET_SEQ.indexOf(name) >= 0 || (!still && !heat && hits >= 2);                                              // RPLK・LPRK
  const seqW = !still && (hits >= 2 || (hits >= 1 && /,特殊/.test(atk)));                   // WPLK など
  return str.replace(SHEET_BTN_RE, (m, off) => {
    const parts = m.split("+");
    let tok;
    if (parts.length === 1) tok = (SHEET_SPLIT[m] && ((m.length === 4 && /^(?:LPRK|RPLK)$/.test(m)) ? seqPair : seqW)) ? SHEET_SPLIT[m] : m;
    else {
      const set = new Set(); let total = 0;
      parts.forEach(p => { SHEET_NUMS[p].split("").forEach(n => set.add(n)); total += SHEET_NUMS[p].length; });
      tok = set.size === total ? SHEET_SPELL[[...set].sort().join("")] : m;   // ボタンが重なる書き方（RP+WP など）は、そのまま
    }
    return tok + (SHEET_BTN_HEAD.test(str.slice(off + m.length)) ? "." : "");
  });
}

/* ヒット・カウンターの列に薄字で出す印。ヒット側＝追撃可能(hf)・コンボ始動(hc)・追撃可能(壁際のみ)(whf)・コンボ始動(壁際のみ)(whc)／カウンター側＝cf・cc・wcf・wcc。
   hf・hc・cf・cc は技名の右のタグと同じ判定（自動＋ t）。壁際のみの4つは t に書く。コンボ始動の薄字は、その欄の答えに「コンボ始動」がある技には出さない（hintItemsOf）。 */
const HINT_ITEMS = {   // 文字の「|」は、画面が狭いときに折り返してよい位置（表示では消える）
  hit:     [["hf", "追撃可能", ""], ["hc", "コンボ|始動", ""], ["whf", "追撃可能", "(壁際|のみ)"], ["whc", "コンボ|始動", "(壁際|のみ)"]],
  counter: [["cf", "追撃可能", ""], ["cc", "コンボ|始動", ""], ["wcf", "追撃可能", "(壁際|のみ)"], ["wcc", "コンボ|始動", "(壁際|のみ)"]]
};
function hintIds(raw) {   // その技のタグの印（自動＋ t）に、t に書いた壁際のみの印を足したもの
  const ids = new Set(moveInfo(raw).tags.map(t => t.id));
  const o = (raw && typeof raw === "object") ? raw : {};
  String(o.t || "").split(/[,，\s]+/).forEach(x => { if (!x) return; if (x[0] === "-") ids.delete(x.slice(1)); else ids.add(x); });
  return ids;
}
/* 薄字にする項目。コンボ始動の2種（hc・whc／cc・wcc）は、その欄の答え（a.hit／a.counter）に「コンボ始動」がすでに書いてある技には出さない（答えと二重にしない）。
   つまり出るのは、「コンボ始動」に指定した技のうち、その欄の答えに「コンボ始動」が入っていない技だけ */
function hintItemsOf(raw, side) {
  const ids = hintIds(raw);
  const a = (raw && typeof raw === "object" && raw.a) || {};
  const has = /コンボ始動/.test(String(side === "hit" ? a.hit : a.counter || ""));
  return HINT_ITEMS[side].filter(it => ids.has(it[0]) && !(has && /^w?[hc]c$/.test(it[0])));
}
function moveHints(raw) {
  const pick = side => hintItemsOf(raw, side).map(it => (it[1] + it[2]).replace(/\|/g, ""));
  return { hit: pick("hit"), counter: pick("counter") };
}
/* ヒット・カウンターの列のセルに足す薄字（項目ごとに1行。「(壁際のみ)」は次の行）。出すものがなければ null */
function hintEl(raw, colId) {
  if (colId !== "hit" && colId !== "counter") return null;
  const items = hintItemsOf(raw, colId);
  if (!items.length) return null;
  const put = (el, text) => text.split("|").forEach((part, i) => { if (i) el.appendChild(document.createElement("wbr")); el.appendChild(document.createTextNode(part)); });
  const box = document.createElement("span"); box.className = "chnote"; box.dataset.hint = colId;

  if (/^立ち途中/.test(c)) return { id: "ws", label: "立ち途中", title: "立ち途中（ワンダッシュ後などの立ち上がり中）から出す技" };
  if (/^(?:しゃがみ中|しゃがんだ状態)/.test(c)) return { id: "fc", label: "しゃがみ中", title: "しゃがみ中・しゃがんだ状態から出す技" };
  if (/^横移動中/.test(c)) return { id: "sw", label: "横移動中", title: "横移動中に出す技" };
  if (/^(?:背向け中|相手に背を向けて)/.test(c)) return { id: "bt", label: "背向け中", title: "背向け中（相手に背を向けた状態）から出す技" };
  if (/^壁を背に/.test(c)) return { id: "wl", label: "壁を背にして", title: "壁を背にして出す技" };
  if (/^(?:仰向け|うつ伏せ)ダウン中/.test(c)) return { id: "dn", label: "ダウン中", title: "ダウン中（仰向け・うつ伏せ）に出す技" };
  if (!depth && (m = c.match(/^[^A-Z（(【]{1,16}?(?:状態で|保有時に?)(.+)$/))) return { rest: m[1] };
  if ((m = c.match(/^(.+?中)(?=[0-9nA-Z（(～【]|の|に|で|背向け|$)/))) { const nm = m[1].replace(/\d+(?:～\d+)?発目(?=中$)/, ""); return { id: "st:" + nm, label: nm, title: nm + "から出す技（構え・状態）" }; }
  return { id: "etc", label: "その他", title: "上のどれにも当てはまらない技" };
}
/* 1つの技が入る区分の一覧： [{ id, label, title, group }] */
function jumpKeys(raw) {
  const out = [];
  const o = (raw && typeof raw === "object") ? raw : {};
  const a = o.a || {}, name = o.name || "", cmd = a.command || "";
  const info = moveInfo(raw), ids = new Set(info.tags.map(t => t.id));
  const heatOnly = ["ho", "hb", "hs", "ra"].filter(k => ids.has(k));
  heatOnly.forEach(k => out.push({ id: "h:" + k, group: "heat" }));
  if (ids.has("he")) out.push({ id: "h:he", group: "heat" });
  const heatState = HEAT_STATE_RE.test(cmd) && !HEAT_SMASH_RE.test(cmd);
  if (heatState && !ids.has("ho") && !ids.has("ra")) out.push({ id: /ヒートダッシュ/.test(name) ? "h:hd" : "h:hv", group: "heat" });
  if (heatOnly.length) return out;   // ヒート専用技・バースト・スマッシュ・レイジアーツは、入力方向や構えのほうには入れない
  // 入力方向／状況・構え
  let c = jumpCore(cmd);
  for (let d = 0; d < 3; d++) {
    const m = c.match(/^[0-9n]+/);
    if (m) { const k = jumpDirOf(m[0]); out.push({ id: "d:" + k, group: "dir" }); return out; }
    if (/^[LRW][PKLR]/.test(c)) { out.push({ id: "d:n", group: "dir" }); return out; }
    const s = jumpSitOf(c, d);
    if (s.rest) { c = s.rest; continue; }
    out.push({ id: "s:" + s.id, group: "sit", label: s.label, title: s.title });
    return out;
  }
  out.push({ id: "s:etc", group: "sit", label: "その他", title: "上のどれにも当てはまらない技" });
  return out;
}
/* 区分の見た目の名前など */
function jumpMeta(k) {
  const [g, v] = [k.id.slice(0, k.id.indexOf(":")), k.id.slice(k.id.indexOf(":") + 1)];
  if (g === "d") return { label: JUMP_DIR_LABEL[v] || v, disp: v === "n" ? "n" : (JUMP_DIR_LABEL[v] || v), arrow: true, title: JUMP_DIR_TITLE[v] || (jumpArrows(v) + "（" + JUMP_DIR_NAMES[v] + "）から出す技") };
  if (g === "h") { const h = JUMP_HEAT.find(x => x.id === v); return { label: h.label, disp: h.label, title: h.title }; }
  return { label: k.label, disp: k.label, title: k.title };
}
/* キャラの技の一覧 → 区分ごとの { id, group, label, title, idx:[技の位置…] }（表の上から順） */
function jumpBuild(raws) {
  const map = new Map();
  raws.forEach((raw, idx) => {
    jumpKeys(raw).forEach(k => {
      let s = map.get(k.id);
      if (!s) { const mt = jumpMeta(k); s = { id: k.id, group: k.group, label: mt.label, disp: mt.disp, arrow: !!mt.arrow, title: mt.title, idx: [] }; map.set(k.id, s); }
      s.idx.push(idx);
    });
  });
  const all = [...map.values()];
  const heatOrder = JUMP_HEAT.map(h => "h:" + h.id);
  const secs = [];
  JUMP_GROUPS.forEach(g => {
    let list = all.filter(s => s.group === g.id);
    if (g.id === "heat") list.sort((x, y) => heatOrder.indexOf(x.id) - heatOrder.indexOf(y.id));
    else list.sort((x, y) => x.idx[0] - y.idx[0]);   // 表の上から出てくる順
    list.forEach(s => secs.push(s));
  });
  return secs;
}

/* ---------- 画面（PC版・スマホ版で共通） ----------
   jumpSetup({ host, rowOf(idx)→行の要素 or null, offset()→固定ヘッダーの高さ, collapse→true なら、各段を1行に収めて「ほか○件」で広げる, closeOnGo→true なら、項目を押して飛んだとき広げた段を閉じる（スマホ版） })
   jumpSet(charId, raws)  … キャラが変わったら（同じキャラなら何もしない）
   jumpAvail(visible(idx)→表示中か) … 検索などで絞り込んだとき、該当が無い項目を薄くする */
let jumpCfg = null, jumpCur = null, jumpSecs = [], jumpVis = null;
function jumpSetup(cfg) {
  jumpCfg = cfg;
  if (cfg.collapse && window.ResizeObserver) {   // コンボ集から戻って表示されたときなどに、測り直す
    // 通知の最中に DOM を書き換えると、高さが変わってまた通知が来る（ブラウザが「ResizeObserver loop …」のエラーを出す）。
    // 測り直しは幅が変わったときだけ・次のフレームで行う（段の折り返しは幅だけで決まる）
    let lastW = -1;
    new ResizeObserver(es => {
      const w = Math.round(es[es.length - 1].contentRect.width);
      if (w === lastW) return;
      lastW = w;
      cancelAnimationFrame(jumpFit.raf); jumpFit.raf = requestAnimationFrame(jumpFit);
    }).observe(cfg.host);
  }
}
function jumpSet(cid, raws) {
  if (!jumpCfg) return;
  if (jumpCur === cid && jumpCfg.host.childElementCount) return;
  jumpCur = cid; jumpSecs = jumpBuild(raws || []);
  const host = jumpCfg.host; host.innerHTML = "";
  host.hidden = !jumpSecs.length;
  JUMP_GROUPS.forEach(g => {
    const list = jumpSecs.filter(s => s.group === g.id);
    if (!list.length) return;
    const row = document.createElement("div"); row.className = "jrow"; row.dataset.g = g.id;
    const lb = document.createElement("span"); lb.className = "jlbl"; lb.textContent = g.label; row.appendChild(lb);
    const ch = document.createElement("div"); ch.className = "jchips";
    list.forEach(s => {
      const b = document.createElement("button"); b.type = "button"; b.className = "jchip"; b.dataset.id = s.id;
      b.title = s.title + "：" + s.idx.length + "件　（押すと、最初の技へ移動します）";
      b.setAttribute("aria-label", s.label + "の最初の技へ移動（" + s.idx.length + "件）");
      const t = document.createElement("span"); t.className = "jt" + (s.arrow ? " ja" : ""); if (s.arrow) jumpArrowFill(t, s.disp); else t.textContent = s.disp;
      const n = document.createElement("span"); n.className = "jn"; n.textContent = s.idx.length;
      b.append(t, n);
      b.addEventListener("click", () => {
        if (jumpCfg.closeOnGo && row.classList.contains("open")) { row.classList.remove("open"); jumpFit(); }   // 先に閉じてから飛ぶ（閉じると高さが変わるので、飛ぶ位置は閉じたあとで測る）
        jumpGo(s.id);
      });
      ch.appendChild(b);
    });
    row.appendChild(ch);
    if (jumpCfg.collapse) {
      const mo = document.createElement("button"); mo.type = "button"; mo.className = "jmore"; mo.hidden = true;
      mo.addEventListener("click", () => { row.classList.toggle("open"); jumpFit(); });
      row.appendChild(mo);
    }
    host.appendChild(row);
  });
  if (jumpVis) jumpAvail(jumpVis);
  jumpFit();
}
/* 1行に収まらない段に「すべて表示」を出す（PC版）。隠れている間は測れないので、表示されたときにやり直す */
function jumpFit() {
  if (!jumpCfg || !jumpCfg.collapse) return;
  jumpCfg.host.querySelectorAll(".jrow").forEach(row => {
    const ch = row.querySelector(".jchips"), mo = row.querySelector(".jmore");
    if (!ch.getClientRects().length) return;
    const open = row.classList.contains("open");
    const hidden = open ? 0 : [...ch.children].filter(b => b.offsetTop > 10).length;   // 1行目に入りきらなかった項目の数
    const wrapped = open ? [...ch.children].some(b => b.offsetTop > 10) : hidden > 0;
    mo.hidden = !wrapped;
    mo.textContent = open ? "たたむ ▲" : "ほか" + hidden + "件 ▼";
    mo.setAttribute("aria-expanded", String(open));
  });
}

if (document.fonts && document.fonts.ready) document.fonts.ready.then(jumpFit);

/* 行・カード（el）を入れている「縦にスクロールする枠」を返す（無ければ null）。
   PC版は、窓の幅が1000px以下だと、表を枠の中で縦にも横にもスクロールさせる（枠の高さは画面の80%）。そのとき行はページではなく枠の中を動くので、ページだけを動かしても行へ飛べない。
   スマホ版（カードがページに並ぶ）や、幅1001px以上のPC版（表がページに並ぶ）では、見つからない（null）。 */
function jumpScroller(el) {
  for (let e = el && el.parentElement; e && e !== document.body && e !== document.documentElement; e = e.parentElement) {
    const o = getComputedStyle(e).overflowY;
    if ((o === "auto" || o === "scroll") && e.scrollHeight > e.clientHeight + 1) return e;
  }
  return null;
}
/* 技の位置（idx）の行・カードまで画面を動かして、光らせる。onEl(要素) があれば、光らせたあとに呼ぶ（数字ジャンプが、該当のセルにも印を付けるのに使う）。描画の途中で行がまだ無いときは、少し待ってやり直す */
function jumpShow(idx, tries, onEl) {
  if (!jumpCfg) return;
  const el = jumpCfg.rowOf(idx);
  if (!el) { if ((tries || 0) < 30) requestAnimationFrame(() => jumpShow(idx, (tries || 0) + 1, onEl)); return; }
  const calm = window.matchMedia && matchMedia("(prefers-reduced-motion: reduce)").matches, beh = calm ? "auto" : "smooth";
  const sc = jumpScroller(el);
  if (sc) {
    /* 表が枠の中で縦にスクロールする配置（PC版・窓の幅が1000px以下）：行はページではなく枠の中を動くので、枠を動かして行を見出しの下に出し、ページは枠の上端が画面の上（貼りついた数字ジャンプの帯の下）に来る位置へ動かす（すでにそこなら動かない） */
    const th = sc.querySelector("thead th"), hh = th ? th.getBoundingClientRect().height : 0;
    const rt = el.getBoundingClientRect().top - sc.getBoundingClientRect().top + sc.scrollTop;
    sc.scrollTo({ top: Math.max(0, rt - hh - 6), behavior: beh });
    const bar = typeof njBarH === "function" ? njBarH() : 0;
    window.scrollTo({ top: Math.max(0, sc.getBoundingClientRect().top + window.pageYOffset - bar - 6), behavior: beh });
  } else {
    const top = el.getBoundingClientRect().top + window.pageYOffset - jumpCfg.offset() - 6;
    window.scrollTo({ top: Math.max(0, top), behavior: beh });
  }
  document.querySelectorAll(".jflash").forEach(e => e.classList.remove("jflash"));   // 前の印は消す
  void el.offsetWidth; el.classList.add("jflash");
  clearTimeout(jumpGo.t); jumpGo.t = setTimeout(() => el.classList.remove("jflash"), 1900);
  if (onEl) onEl(el);
}
function jumpGo(id) {
  const s = jumpSecs.find(x => x.id === id); if (!s || !jumpCfg) return;
  const idx = jumpVis ? s.idx.find(i => jumpVis(i)) : s.idx[0];
  if (idx === undefined) return;
  jumpShow(idx, 0);
}

/* ブックマークと、数値での絞り込み・並べ替え（パソコン版・スマホ版で共通の部品）
   画面への組み込み（行・カードの作り方、一覧の並べ替え方）は各版で行い、ここには判定・保存・パネルの部品だけを置く。
   ■ ブックマーク：技ごとの旗。チェックとは別のキー（tekken-checksheet-review-v1）に { "キャラid:技ID": true } で保存する。
   ■ 数値の絞り込み・並べ替え（NF）：その場かぎりの設定で、保存はしない。
     ヒット・カウンターの状態（ダウン・浮き・受身・コンボ始動・倒れ方）での絞り込みも同じ NF に入れる（NF.states）。状態の判定は、答えの文字（「叩きつけダウン⏎コンボ始動、受身不可」など）に対する ST_GROUPS の判定。
     比べる数字の取り方（数字のない値は「該当なし」＝絞り込み中は出さず、並べ替えでは最後）：
       発生・ガード・ヒット・カウンター・全体硬直 … 最初の数字（+8 → 8、-12s → -12、±0 → 0、10F → 10。「D」「ダウン」「-」などは数字なし）
       ※ガード・ヒット・カウンターのマイナスの数字は、格ゲーの言い方：「-10以上」＝ -10・-11・-12…（nfOp を参照）
       ダメージ・削りダメージ … カンマ区切りの合計（5,8,18 → 31、3×7 → 21、( ) の中は数えない、「40 or 45」「20～180」は最初の数字）
       対象外（斜線）の項目は数字なし */

/* ---------- ブックマーク ---------- */
const REVIEW_KEY = "tekken-checksheet-review-v1";
let reviews = {};   // { "キャラid:技ID": true }
try { const o = JSON.parse(localStorage.getItem(REVIEW_KEY)); if (o && typeof o === "object" && !Array.isArray(o)) reviews = o; } catch (e) { reviews = {}; }
const rvKey = (cid, k) => cid + ":" + k;
function rvSave() { try { localStorage.setItem(REVIEW_KEY, JSON.stringify(reviews)); } catch (e) {} }
const rvHas = (cid, k) => reviews[rvKey(cid, k)] === true;
function rvToggle(cid, k) {
  const kk = rvKey(cid, k);
  if (reviews[kk]) delete reviews[kk]; else reviews[kk] = true;
  rvSave();
  return !!reviews[kk];
}
function rvCount(c) { let n = 0; c.ks.forEach(k => { if (reviews[rvKey(c.id, k)] === true) n++; }); return n; }
function rvClearChar(cid) { Object.keys(reviews).forEach(kk => { if (kk.indexOf(cid + ":") === 0) delete reviews[kk]; }); }
/* 書き出し用：{ キャラid: [技ID, …] } */
function rvExport() {
  const r = {};
  Object.keys(reviews).forEach(kk => {
    const p = kk.split(":");
    if (p.length !== 2 || reviews[kk] !== true || !/^\d+$/.test(p[1])) return;
    (r[p[0]] = r[p[0]] || []).push(+p[1]);
  });
  Object.keys(r).forEach(id => r[id].sort((a, b) => a - b));
  return r;
}
/* 読み込み用：ids に入っているキャラのブックマークを、r の内容で置き換える（r に無いキャラは、すべて外れる）。付けた数を返す */
function rvImport(r, ids, chars) {
  let n = 0;
  ids.forEach(id => {
    if (!chars[id]) return;
    rvClearChar(id);
    (Array.isArray(r[id]) ? r[id] : []).forEach(k => {
      if (Number.isInteger(k) && chars[id].ks.has(k) && !reviews[rvKey(id, k)]) { reviews[rvKey(id, k)] = true; n++; }
    });
  });
  rvSave();
  return n;
}
/* 旗のボタン（技の行・カードに付ける）。押すと付け外しして、after(付いたか, ボタン) を呼ぶ */
function rvBtnEl(cid, k, name, after) {
  const b = document.createElement("button");
  b.type = "button"; b.className = "rvbtn";
  b.innerHTML = '<svg viewBox="0 0 24 24" aria-hidden="true" focusable="false"><path d="M6 21V4"/><path d="M6 4h12l-3 4.5L18 13H6z"/></svg>';
  const sync = () => {
    const on = rvHas(cid, k);
    b.setAttribute("aria-pressed", String(on));
    b.setAttribute("aria-label", name + (on ? "：ブックマーク（押すと外す）" : "：ブックマークにする"));

  };
  sync();
  b.addEventListener("click", e => {
    e.stopPropagation(); e.preventDefault();
    const on = rvToggle(cid, k);
    sync();
    if (after) after(on, b);
  });
  return b;
}

/* ---------- 数値の読み取り ---------- */
const NUM_COLS = [
  { id: "startup",  label: "発生",           unit: "F" },
  { id: "damage",   label: "ダメージ",       unit: "", sum: true },
  { id: "guard",    label: "ガード状況",     unit: "F", adv: true },
  { id: "hit",      label: "ヒット状況",     unit: "F", adv: true },
  { id: "counter",  label: "カウンター状況", unit: "F", adv: true },
  { id: "recovery", label: "全体硬直",       unit: "F" },
  { id: "chip",     label: "削りダメージ",   unit: "", sum: true }
];
const NUM_COLMAP = Object.fromEntries(NUM_COLS.map(c => [c.id, c]));
function numClean(t) {
  return String(t == null ? "" : t)
    .replace(/[０-９]/g, ch => String.fromCharCode(ch.charCodeAt(0) - 0xFEE0))
    .replace(/[−－ー‐‑–—―]/g, "-").replace(/＋/g, "+").replace(/[×＊]/g, "*").replace(/[，、]/g, ",").trim();
}
/* 最初の数字（符号つき）。数字で始まらない値（D・ダウン・浮・投げ・-）は null */
function numFrame(t) {
  const m = /^([+\-±]?)\s*(\d+)/.exec(numClean(t));
  if (!m) return null;
  return m[1] === "-" ? -(+m[2]) : +m[2];
}
/* カンマ区切りの合計（3*4 や 3×4 は 3 を4回）。( ) の中は数えない。数字が1つも無ければ null */
function numSum(t) {
  const s = numClean(t).replace(/\([^)]*\)|（[^）]*）/g, "");
  let total = 0, found = false;
  s.split(",").forEach(p => {
    const m = /^\s*(-?\d+)\s*(?:\*\s*(\d+))?/.exec(p);
    if (!m) return;
    total += +m[1] * (m[2] ? +m[2] : 1); found = true;
  });
  return found ? total : null;
}
/* 技1つぶんの数字：{ startup, damage, guard, … }（読み取れないものは null） */
function moveNums(raw, ch) {
  const o = (raw && typeof raw === "object") ? raw : {};
  const a = o.a || {}, off = o.off || [];
  const r = {};
  NUM_COLS.forEach(c => {
    const t = a[c.id];
    r[c.id] = (off.indexOf(c.id) >= 0 || t == null || t === "") ? null : (c.sum ? numSum(t) : numFrame(t));
  });
  /* 壁やられの絞り込み：キャラに wallTable が付いている（壁やられ欄を表で確かめて登録した）ときは、ヒット欄・カウンター欄の文字ではなく、壁やられ欄の答えで判定する。
     答えは「壁やられ(強)」「壁やられ(中)」「壁やられ(強・中)」、またはヒットとカウンターで違うときは「ヒット：…⏎カウンター：…」（片方だけなら「ヒット：…」「カウンター：…」の1行）。
     hitW・counterW は、その側で誘発する壁やられの印（壁強・壁中）。null のキャラは、従来どおり文字から判定する。 */
  const wl = String(a.wall || ""), wh1 = /^ヒット：([^\n]*)$/.exec(wl), wc1 = /^カウンター：([^\n]*)$/.exec(wl);   // 「ヒット：…」だけ／「カウンター：…」だけの1行の答えは、もう一方の側は誘発なし
  const wsp = /^ヒット：([^\n]*)\nカウンター：([^\n]*)$/.exec(wl) || (wh1 ? [wl, wh1[1], ""] : wc1 ? [wl, "", wc1[1]] : null);
  const wmk = x => (/壁やられ\(強/.test(x) ? "壁強" : "") + (/中\)/.test(x) ? "壁中" : "");
  const wok = !!(ch && ch.wallTable) && off.indexOf("wall") < 0 && wl && !/^ガード：/.test(wl);
  r.hitW = (ch && ch.wallTable) ? (wok ? wmk(wsp ? wsp[1] : wl) : "") : null;
  r.counterW = (ch && ch.wallTable) ? (wok ? wmk(wsp ? wsp[2] : wl) : "") : null;
  ["hit", "counter"].forEach(id => { const t = a[id]; r[id + "T"] = (off.indexOf(id) >= 0 || t == null || t === "") ? null : String(t); });   // 状態の絞り込み用（答えの文字そのまま）
  return r;
}
/* 入力欄の文字 → 整数（全角・「−」・末尾の F も受け付ける）。読めなければ null */
function parseNumInput(t) {
  const s = numClean(t).replace(/[FfＦｆ]$/, "").replace(/^\+/, "").trim();
  return /^-?\d+$/.test(s) ? parseInt(s, 10) : null;
}

function nfStatePass(c, nums) {
  const n = nums || {};
  if (c.col === "hit") return nfStateOne(c.key, n.hitT, n.hitW);
  if (c.col === "counter") return nfStateOne(c.key, n.counterT, n.counterW);
  return nfStateOne(c.key, n.hitT, n.hitW) && nfStateOne(c.key, n.counterT, n.counterW);
}
/* 選ぶときに添える「このキャラの該当数」 */
function nfStateCount(col, key, numsList) { const c = { col, key }; let n = 0; numsList.forEach(x => { if (nfStatePass(c, x)) n++; }); return n; }
/* ガード・ヒット・カウンター（有利・不利フレーム）は、格ゲーの言い方に合わせる：マイナスの数字では「以上」が「マイナスが多い（-11、-12…）」、「以下」が「マイナスが少ない」の意味。
   例）ガード -10以上 ＝ -10・-11・-12…（数の大小では ≤ -10）／ガード -10以下 ＝ -10・-9…・±0・+1…（数の大小では ≥ -10）。プラスの数字（0 を含む）は、そのまま数の大小。 */
function nfOp(c) {
  const k = NUM_COLMAP[c.col];
  if (k && k.adv && c.val < 0 && c.op !== "eq") return c.op === "le" ? "ge" : "le";
  return c.op;
}
function nfPass(nums) {
  const n = nums || {};
  if (!nfActiveStates().every(c => nfStatePass(c, n))) return false;
  return nfActiveConds().every(c => {
    const v = n[c.col];
    if (v === null || v === undefined) return false;
    const op = nfOp(c);
    return op === "le" ? v <= c.val : op === "ge" ? v >= c.val : v === c.val;
  });
}
/* list を並べ替えた新しい配列にする。並べ替えなしのときは ordOf（元の順）で並べる。数字のない技は最後、同じ数字は元の順 */
function nfSorted(list, numsOf, ordOf) {
  const col = NF.sortCol, dir = NF.sortDir === "desc" ? -1 : 1;
  return list.slice().sort((x, y) => {
    if (col) {
      const a = numsOf(x)[col], b = numsOf(y)[col];
      const an = a === null || a === undefined, bn = b === null || b === undefined;
      if (an !== bn) return an ? 1 : -1;
      if (!an && a !== b) return (a - b) * dir;
    }
    return ordOf(x) - ordOf(y);
  });
}

}

   対象は、技表に出ている文字すべて：技名・ふりがな・各項目（COLS：コマンド・攻撃判定・発生・ダメージ・ガード・ヒット・カウンター・壁やられ・ギミック・投げ抜け・削り・ホーミング・追尾方向・ジャンプ・しゃがみ・ヒート消費・全体硬直）の答え・
   コマンドの別表記・タグ（カプセル）と、その呼び名・メモ。「上」「中」「下」のような1文字でも、攻撃判定の文字に当たる。
   項目の境目をまたいで一致しないよう、項目の間に区切り（WS_SEP）を入れる（normQ は空白だけを消すので、区切りは残る）。
   数字だけの検索（数字ジャンプ）は、これまでどおり「技名・コマンドなど」の文字（q）を使う。ここで作る文字（qa）は、数字だけではない検索の絞り込みと色塗りに使う。
     wsCols(raw) … { 項目id: そろえた答えの文字 }（対象外の項目・答えの無い項目は入れない）。色塗りで、どのセルに当たったかを調べるのに使う
     wsHay(raw, memo) … 技1つぶんの、検索対象の文字すべて（そろえ済み） */
const WS_SEP = "\u001f";
function wsCols(raw) {
  const o = (raw && typeof raw === "object") ? raw : {}, a = o.a || {}, off = o.off || [], r = {};
  COLS.forEach(c => {
    if (off.indexOf(c.id) >= 0) return;
    let t = a[c.id]; if (t === null || t === undefined || t === "") return;
    t = String(t).replace(/\n/g, " ");
    if (c.id === "command" && a.ci) t += WS_SEP + String(a.ci).replace(/[.^]/g, "");   // コマンドの別表記（アイコン用の書き方）も対象
    r[c.id] = normQ(t);
  });
  return r;
}
function wsHay(raw, memo) {
  const o = (raw && typeof raw === "object") ? raw : {}, cols = wsCols(raw);
  return normQ([typeof raw === "string" ? raw : (o.name || ""), o.y || ""].concat(Object.keys(cols).map(id => cols[id]), [moveInfo(raw).kw, memo || ""]).join(WS_SEP));
}

/* ---------- 数字ジャンプ（検索欄に数字だけを入れたとき） ----------

   その数字が入っている項目（発生・ダメージ・ガード・ヒット・カウンター・全体硬直・削り）ごとに「ジャンプリンク」を検索欄の下に出す。
   押すとその項目が一致する最初の技へ移動して光り（パソコン版は該当のセルにも印）、もう一度押すと次の技へ進む（件数の表示が 2/5 のように変わる）。
   ・符号つき（-10・+8・±0）は、ガード・ヒット・カウンターだけが対象。符号なし（15）は、全部の項目が対象。
   ・数字の読み取りは numFrame / numSum と同じ（最初の数字）。表示中の技（ブックマーク・数値の絞り込みを通ったもの）だけを数える。
   ・符号なしの数字は、技名・コマンドなどに入っている場合（「66」など）も1つのリンクにする。
     「その文字を含む技だけ表示」のボタンで、今までどおりの文字での絞り込みに切り替えられる（検索の文字を変えると、切り替えは元に戻る）。
   ・数字だけでない検索（66RP・技名など）は、今までどおり絞り込みで、ジャンプリンクは出さない。
   組み込み：numJumpSetup({ host, cellOf(行の要素, 項目id)→セル or null, refilter() })／検索・絞り込みのたびに numMode(入力欄の文字) で

function numQuery(t) {
  const s = numClean(t).replace(/\s+/g, "").replace(/[FfＦｆ]$/, "");
  const m = /^([+\-±]?)(\d{1,4})$/.exec(s);
  if (!m) return null;
  const n = +m[2];
  return { sign: m[1], val: m[1] === "-" ? -n : n, text: m[1] + n };
}

/* 全角・空白をそろえた文字（normQ と同じ）と、元の文字の位置の対応 */

function shMarkText(el, tq) {
  const orig = el.dataset.shOrig !== undefined ? el.dataset.shOrig : el.textContent;
  const ch = shNormChars(orig), ns = ch.map(x => x.c).join("");
  const ranges = []; let from = 0;
  for (;;) { const p = ns.indexOf(tq, from); if (p < 0) break; ranges.push([ch[p].i, ch[p + tq.length - 1].i + 1]); from = p + tq.length; }
  if (!ranges.length) return false;
  el.dataset.shOrig = orig;
  const frag = document.createDocumentFragment(); let pos = 0;
  ranges.forEach(([a, b]) => {
    if (a > pos) frag.appendChild(document.createTextNode(orig.slice(pos, a)));
    const m = document.createElement("mark"); m.className = "shl"; m.textContent = orig.slice(a, b); frag.appendChild(m); pos = b;
  });
  if (pos < orig.length) frag.appendChild(document.createTextNode(orig.slice(pos)));
  el.textContent = ""; el.appendChild(frag);
  return true;
}

function shApply() {
  const cfg = SH.cfg; if (!cfg) return;
  shClear();
  const raw = document.getElementById("q").value || "";
  const tq = normQ(raw), nq = numQuery(raw);
  SH.hits = [];
  if (tq) {
    cfg.rows().forEach(el => {
      const f = cfg.info(el); if (!f) return;
      const numIds = nq ? NUM_COLS.filter(c => !NJ.off[c.id] && (!nq.sign || c.adv) && f.nums && f.nums[c.id] === nq.val).map(c => c.id) : [];   // 「✖」で解除した項目は、色を塗らない
      const hay = nq ? f.hay : (f.qa || f.hay);   // 数字だけの検索は「技名・コマンドなど」の文字／それ以外（ワード検索）は技表の文字すべて
      const textHit = !!hay && hay.includes(tq) && !(nq && NJ.off._t);

      let seen = false;   // 画面に見える印を付けられたか

        if (!nq) {   // ワード検索：攻撃判定・タグの文字と、当たった項目のセル（開いていないカードのセルは無いので、その場合は行の印だけ）
          el.querySelectorAll(".atk, .tg").forEach(n => { if (n.getClientRects().length && shMarkText(n, tq)) seen = true; });
          const cols = f.cols || {};
          Object.keys(cols).forEach(id => {

          });
        }
      }
      numIds.forEach(id => { const c = cfg.cellOf && cfg.cellOf(el, id); if (c) { c.classList.add("nhit"); seen = true; } });
      if (cfg.badge && numIds.length) shBadge(el, numIds, nq);

      SH.hits.push({ el, idx: f.idx });
    });
  }

  clearTimeout(SH.later); SH.later = setTimeout(shMarkers, 400);   // 描画・収め直しが落ち着いてから、もう一度位置を測る
}
/* 右端の目印：ヒットした行の、ページ全体の中での位置（画面の高さに縮めて並べる）。近い位置は1本にまとめる */
function shMarkers() {
  const strip = SH.strip; if (!strip) return;
  strip.textContent = "";
  const H0 = Math.max(document.documentElement.scrollHeight, 1), vh = window.innerHeight;
  /* 行の位置（0〜1）：ふつうはページ全体の長さに対する位置。表が枠の中で縦にスクロールするとき（PC版・幅1000px以下）は、枠の中身の長さに対する位置（ページは短いので、ページの長さでは測れない） */
  const pts = SH.hits.filter(h => h.el.isConnected && h.el.getClientRects().length).map(h => {
    const r = h.el.getBoundingClientRect(), sc = typeof jumpScroller === "function" ? jumpScroller(h.el) : null;
    const y = sc ? r.top - sc.getBoundingClientRect().top + sc.scrollTop + r.height / 2 : r.top + window.pageYOffset + r.height / 2;
    return { f: y / (sc ? Math.max(sc.scrollHeight, 1) : H0), idx: h.idx };
  }).sort((a, b) => a.f - b.f);
  if (!pts.length) { strip.hidden = true; return; }
  const ticks = []; let last = -99;
  pts.forEach(p => {
    const k = Math.round(p.f * vh);
    if (k - last >= 4 || !ticks.length) { ticks.push({ k, n: 1, idx: p.idx }); last = k; } else ticks[ticks.length - 1].n++;
  });
  const total = pts.length;
  ticks.forEach(t => {
    const b = document.createElement("button"); b.type = "button"; b.className = "shtick"; b.style.top = Math.max(0, t.k - 1) + "px";
    b.title = "ヒットした技へ移動" + (t.n > 1 ? "（このあたりに" + t.n + "件）" : "") + "／全" + total + "件";
    b.setAttribute("aria-label", b.title);
    b.addEventListener("click", () => { jumpShow(t.idx, 0); });
    strip.appendChild(b);
  });
  strip.hidden = false;
}

/* 答えの表示：「状態名」と「性質」が改行で区切られている答え（例：転ばしダウン（正面）⏎コンボ始動、受身不可）は、
   状態名を大きく・性質を小さく2段で出す。（ ）の部分はひとかたまりで折り返す。1行だけの答え（+14 など）はそのまま */
function setAns(el, text, colId) {
  el.textContent = "";
  const t = text || "";
  if (colId === "gimmick" && t) {      // ギミック：複数あるときは、1つずつのタグにして並べる（床のギミックだけ色を変える）
    const floor = ["フロアブレイク", "ハードフロアブレイク", "フロアブラスト"];
    el.classList.add("gm");
    t.split("\n").filter(Boolean).forEach(n => {
      const cap = n.charAt(0) === "※";     // 「※」で始まる行は、タグではなく小さな注記（床のギミックが発動する条件）
      const s = document.createElement("span"); s.className = cap ? (n.charAt(1) === "壁" ? "gc w" : "gc") : "gi" + (floor.indexOf(n) >= 0 ? " f" : ""); s.textContent = cap ? n.slice(1) : n; el.appendChild(s);
    });
    return;
  }
  if (colId === "startup" && t.indexOf("\n") < 0) {      // 発生(持続)：1段目に発生、2段目に持続（例：10F ⏎ (1F)）。表示だけで、データの文字はそのまま（textContent は元の答えと同じ）
    const k = t.search(/[(（]/);
    if (k > 0) {
      el.classList.add("m2", "st2");
      const a1 = document.createElement("span"); a1.className = "a1"; a1.textContent = t.slice(0, k);
      const a2 = document.createElement("span"); a2.className = "a2"; const rest = t.slice(k);
      const gs = rest.match(/[(（][^)）]*[)）]/g);      // 「(2段目：15F)(3段目：2F)」のように続くときは、かっこのまとまりごとに1つの塊にして、塊の間だけで折り返す
      const brk = (el, s) => s.split("：").forEach((x, n) => { if (n) { el.appendChild(document.createTextNode("：")); el.appendChild(document.createElement("wbr")); } el.appendChild(document.createTextNode(x)); });   // 「段目：」のあとだけで折り返せる
      if (gs && gs.length > 1 && gs.join("") === rest) gs.forEach((g, n) => { if (n) a2.appendChild(document.createElement("wbr")); const s = document.createElement("span"); s.className = "stg"; brk(s, g); a2.appendChild(s); });
      else if (rest.indexOf("：") >= 0) brk(a2, rest);
      else { a2.className = "a2 one"; a2.textContent = rest; }
      el.append(a1, a2); return;
    }
  }
  const wrap = (parent, s) => s.split(/(（[^）]*）)/).forEach((p, i) => {
    if (!p) return;
    if (i % 2) { const n = document.createElement("span"); n.className = "pn"; n.textContent = p; parent.appendChild(n); }
    else parent.appendChild(document.createTextNode(p));
  });
  const i = t.indexOf("\n");
  if (i < 0) { el.textContent = t; return; }
  el.classList.add("m2");
  const a1 = document.createElement("span"); a1.className = "a1"; wrap(a1, t.slice(0, i));
  const a2 = document.createElement("span"); a2.className = "a2"; wrap(a2, t.slice(i + 1));
  el.append(a1, "\n", a2);   // textContent は元の答え（改行つき）と同じになる
}

function togglePop(td, colLabel, answer, colId) {
  if (popCell === td) { hidePop(); return; }
  if (popCell) popCell.classList.remove("peek");
  popCell = td;
  td.classList.add("peek");

  pop.innerHTML = "";
  const q = document.createElement("div");

  const a = document.createElement("div");
  a.className = "a" + (answer ? "" : " empty");
  setAns(a, answer || "未登録", colId);
  pop.append(q, a);

  // いったん非表示のまま測定して位置を決める
  // 表示中でも演出をやり直せるよう、アニメーションを止めて初期状態へ即リセット
  pop.style.transition = "none";
  pop.classList.remove("show");
  pop.style.left = "0px"; pop.style.top = "0px";
  const r = td.getBoundingClientRect();
  const w = pop.offsetWidth, h = pop.offsetHeight;
  let left = r.left + r.width / 2 - w / 2;
  left = Math.max(8, Math.min(left, window.innerWidth - w - 8));
  let top = r.bottom + 6, below = true;
  if (top + h > window.innerHeight - 8) { top = r.top - h - 6; below = false; }
  pop.style.left = left + "px";
  pop.style.top = top + "px";
  pop.style.setProperty("--ox", (r.left + r.width / 2 - left) + "px");
  pop.style.setProperty("--oy", below ? "0%" : "100%");
  pop.style.setProperty("--rise", below ? "-10px" : "10px");
  void pop.offsetWidth;            // 初期状態を確定させる
  pop.style.transition = "";       // アニメーションを戻す
  void pop.offsetWidth;
  pop.classList.add("show");
}

/* ---------- 書き出し・読み込み（パソコン版・スマホ版で共通のコード形式 TKCS2） ----------
   チェックとメモをまとめて移せる。旧形式 TKCS1（チェックのみ）も読み込める。
   コードに含まれるキャラのチェック／メモだけを上書きし、含まれないキャラは触らない。 */
function ioExportCode() {
  const s = {}, m = {};
  Object.keys(state).forEach(k => {
    const p = k.split(":"); if (p.length !== 3) return;
    (s[p[0]] = s[p[0]] || []).push(p[1] + ":" + p[2]);
  });
  Object.keys(memos).forEach(k => {
    const p = k.split(":"); if (p.length !== 3 || p[2] !== "memo" || !memos[k]) return;
    (m[p[0]] = m[p[0]] || {})[p[1]] = memos[k];
  });
  return "TKCS2:" + JSON.stringify({ s, m, r: rvExport() });
}
function ioImportCode(raw) {
  raw = String(raw).trim();
  let s = {}, m = {}, r = null;
  if (raw.startsWith("TKCS2:")) { const o = JSON.parse(raw.slice(6)); s = o.s || {}; m = o.m || {}; if (o.r && typeof o.r === "object" && !Array.isArray(o.r)) r = o.r; }   // r（ブックマーク）が無い古いコードでは、ブックマークは触らない
  else if (raw.startsWith("TKCS1:")) { s = JSON.parse(raw.slice(6)); }
  else throw new Error("コードの形式が正しくありません");
  const chars = {};
  CHARACTERS.forEach(c => { chars[c.id] = c; });
  const ids = new Set(Object.keys(s).concat(Object.keys(m), r ? Object.keys(r) : []).filter(id => chars[id]));
  if (!ids.size) throw new Error("読み込めるキャラのデータがありません");
  let nMemo = 0;
  Object.keys(s).forEach(id => {
    if (!chars[id]) return;
    Object.keys(state).forEach(k => { if (k.indexOf(id + ":") === 0) delete state[k]; });
    (Array.isArray(s[id]) ? s[id] : []).forEach(x => {
      if (typeof x === "string" && /^\d+:[a-z]+$/.test(x) && chars[id].ks.has(+x.split(":")[0])) state[id + ":" + x] = true;
    });
  });
  Object.keys(m).forEach(id => {
    if (!chars[id]) return;
    Object.keys(memos).forEach(k => { if (k.indexOf(id + ":") === 0) delete memos[k]; });
    const mm = m[id] || {};
    Object.keys(mm).forEach(i => {
      const t = mm[i];
      if (/^\d+$/.test(i) && chars[id].ks.has(+i) && typeof t === "string" && t) { memos[id + ":" + i + ":memo"] = t.slice(0, 1000); nMemo++; }
    });
  });
  const nRv = r ? rvImport(r, ids, chars) : 0;
  return { chars: ids.size, memos: nMemo, reviews: nRv };
}

}
